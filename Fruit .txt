local function MQRTSTE_fake_script() -- DraggableFrame.LocalScript 
	local script = Instance.new('LocalScript', DraggableFrame)

	local frame = script.Parent
	local closeButton = frame:FindFirstChild("CloseButton")
	local UserInputService = game:GetService("UserInputService")
	
	-- Bi?n luu tr?ng th�i k�o th?
	local dragging = false
	local dragStart = nil
	local startPos = nil
	local lastInput = nil
	
	-- X? l� khi b?t d?u k�o
	local function onInputBegan(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = frame.Position
			lastInput = input
		end
	end
	
local function onInputChanged(input)
		if dragging and input == lastInput then
			local delta = input.Position - dragStart
			frame.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + delta.X,
				startPos.Y.Scale, startPos.Y.Offset + delta.Y
			)
		end
	end
	
	local function onInputEnded(input)
		if input == lastInput then
			dragging = false
		end
	end
	
	frame.InputBegan:Connect(onInputBegan)
	UserInputService.InputChanged:Connect(onInputChanged)
	UserInputService.InputEnded:Connect(onInputEnded)
	
	if closeButton then
		closeButton.MouseButton1Click:Connect(function()
			frame.Visible = false
		end)
	end
	
end
coroutine.wrap(MQRTSTE_fake_script)()

local function GetFruitList()
    local fruitList = {}

    for _, v in ipairs(game.Workspace:GetDescendants()) do
        if v:IsA("Model") and v:FindFirstChild("Name") and string.find(v.Name, "Fruit") then
            table.insert(fruitList, "🍎 " .. v.Name)
        end
    end

    return #fruitList > 0 and table.concat(fruitList, "\n") or "No fruit found."
end

function PostWebhook(Url, message)
    local request = http_request or request or HttpPost or syn.request
    local data =
        request(
        {
            Url = Url,
            Method = "POST",
            Headers = {["Content-Type"] = "application/json"},
            Body = game:GetService("HttpService"):JSONEncode(message)
        }
    )
    return ""
end
